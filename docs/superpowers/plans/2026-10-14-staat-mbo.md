# Staat van de Onderwijsinstelling MBO — implementatieplan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Een mbo-instelling uploadt haar RO/TBG-i-bestanden (of de uitvoer van `mbo-bekostiging-bestanden`), klikt "Samenvoegen", ziet of het gelukt is, klikt "Verder" en krijgt het `staat1cho`-dashboard. Dezelfde stappen werken via de commandoregel.

**Architecture:** Python-pakket `staat-mbo` (import `staat_mbo`, commando `smbo`) gebruikt `mbo-bekostiging-bestanden` om de star te bouwen, controleert de stapel en zet de star om naar een 1CHO-CSV. `staat1cho` (R) krijgt in een aparte PR een optionele voorstap in het dashboard en instelbare diplomasoorten; `smbo dashboard` en `smbo staat` roepen R aan via `Rscript`.

**Tech Stack:** Python ≥ 3.13, polars, uv, pytest, ruff; R ≥ 4.1 met `staat1cho` ≥ 0.3.0, shiny, testthat; Playwright + piper-tts + FFmpeg voor de video.

**Spec:** `docs/superpowers/specs/2026-10-14-staat-mbo-design.md`

## Global Constraints

- Python ≥ 3.13; dependency `mbo-bekostiging-bestanden` vastgepind op commit `05d5c5be33a68b1f7b7a53c57df22cb3a46a3063` (main, 14-10-2026).
- `staat1cho` ≥ 0.3.0 (de versie uit Task 6).

```python
"""Invoer inlezen: ruwe RO/TBG-i-bestanden of de star van mbo-bekostiging-bestanden."""

import os
import zipfile
from pathlib import Path

import polars as pl
from mbo_bekostiging_bestanden.pipeline import (
    detect_bestandstype,
    run_auto_pipeline,
    run_star,
)

VEREISTE_TABELLEN = (
    "fact_inschrijving_schooljaar",
    "fact_inschrijving",
    "dim_deelnemer",
    "dim_opleiding",
    "dim_instelling",
    "meta_leveringen",
)
_SALT = "MBO_PSEUDONIMISERING_SALT"


class InvoerFout(Exception):
    """Fout in de invoer, met een tekst voor de gebruiker."""


def _bestanden(invoer: list[Path], werkmap: Path) -> list[Path]:
    """Alle bestanden uit de invoer: mappen doorzocht, zips uitgepakt."""
    gevonden: list[Path] = []
    for pad in invoer:
        if not pad.exists():
            raise InvoerFout(f"Bestand of map niet gevonden: {pad}")
        if pad.is_dir():
            gevonden += [p for p in sorted(pad.rglob("*")) if p.is_file()]
        elif pad.suffix.lower() == ".zip":
            doel = werkmap / "zip" / pad.stem
            with zipfile.ZipFile(pad) as z:
                z.extractall(doel)
            gevonden += [p for p in sorted(doel.rglob("*")) if p.is_file()]
        else:
            gevonden.append(pad)
    return gevonden


def _lees_parquet(bestanden: list[Path]) -> dict[str, pl.DataFrame]:
    star = {p.stem: pl.read_parquet(p) for p in bestanden if p.suffix == ".parquet"}
    ontbrekend = [t for t in VEREISTE_TABELLEN if t not in star]
    if ontbrekend:
        raise InvoerFout(
            "Het datamodel is niet compleet. Ontbreekt: "
            + ", ".join(ontbrekend)
            + ". Gebruik de volledige uitvoer van mbo-bekostiging-bestanden."
        )
    return star


def _bouw_star(bestanden: list[Path], werkmap: Path) -> dict[str, pl.DataFrame]:
    for b in bestanden:
        soort = detect_bestandstype(b)
        if soort == "grondslag":
            raise InvoerFout(
                f"{b.name}: GRONDSLAG-bestanden gebruiken we niet. Upload alleen RO en TBG-i."
            )
        if soort is None:
            raise InvoerFout(
                f"{b.name}: onbekend bestand. De naam moet beginnen met RO_ of TBGI_."
            )
    if not any(detect_bestandstype(b) == "ro" for b in bestanden):
        raise InvoerFout("Geen RO-bestand gevonden. Upload minstens één RO.")
    if not os.environ.get(_SALT):
        raise InvoerFout(
            f"De sleutel voor pseudonimiseren ontbreekt. Zet de omgevingsvariabele "
            f"{_SALT} (zie README, stap Installeren)."
        )
    brondata = []
    for b in bestanden:
        doel = werkmap / "brondata" / b.stem
        run_auto_pipeline(b, doel)
        brondata.append(doel)
    run_star(brondata, werkmap / "star", scenario="smbo")
    return _lees_parquet(sorted((werkmap / "star" / "datamodel").glob("*.parquet")))


def laad_star(invoer: list[Path], werkmap: Path) -> dict[str, pl.DataFrame]:
    """Lees de invoer als star-tabellen.

    Args:
        invoer:  Ruwe RO/TBG-i-bestanden, een ``datamodel/``-map, losse
                 parquetbestanden of een zip met een van beide.
        werkmap: Map voor tussenbestanden.
    """
    werkmap.mkdir(parents=True, exist_ok=True)
    bestanden = _bestanden(invoer, werkmap)
    if not bestanden:
        raise InvoerFout("Geen bestanden gevonden.")
    if any(b.stem == "fact_inschrijving_schooljaar" for b in bestanden):
        return _lees_parquet(bestanden)
    return _bouw_star(bestanden, werkmap)
```

Let op: `run_auto_pipeline` en `run_star` kunnen `KwaliteitsFout` werpen. Vang die in `_bouw_star` en zet om naar `InvoerFout(f"{b.name}: kwaliteitscontrole mislukt. {e}")`, zodat de gebruiker het bestand ziet. Importeer `KwaliteitsFout` vanuit de module waar `pipeline.py` hem vandaan haalt (`grep -n KwaliteitsFout src/mbo_bekostiging_bestanden/pipeline.py`).

```python
"""Controle op de gestapelde leveringen: dekking, gaten en meldingen."""

import json
from dataclasses import asdict, dataclass, field
from datetime import timedelta

import polars as pl

_MIN_JAREN = 5


@dataclass
class Stapelrapport:
    brins: list[str]
    instellingen: list[str]
    schooljaren: list[int]
    aantal_studenten: int
    gaten: list[dict] = field(default_factory=list)
    fouten: list[str] = field(default_factory=list)
    waarschuwingen: list[str] = field(default_factory=list)

    @property
    def ok(self) -> bool:
        return not self.fouten

    def meldingen(self) -> list[str]:
        """Regels voor de gebruiker; de eerste zegt of het gelukt is."""
        if not self.ok:
            return ["Niet gelukt.", *self.fouten]
        jaren = (
            f"{self.schooljaren[0]}–{self.schooljaren[-1]}"
            if self.schooljaren
            else "geen"
        )
        return [
            "Gelukt: de bestanden zijn samengevoegd tot één bestand.",
            f"Instellingen (BRIN): {', '.join(self.brins)}",
            f"Schooljaren: {jaren} ({len(self.schooljaren)} jaar)",
            f"Studenten: {self.aantal_studenten}",
            *(f"Let op: {w}" for w in self.waarschuwingen),
        ]

    def naar_json(self) -> str:

def _gaten(leveringen: pl.DataFrame) -> list[dict]:

        "BRIN", "DatumBeginPeriode"
    )
    gaten = []

        rijen = groep.select("DatumBeginPeriode", "DatumEindePeriode").rows()
        for (_, einde), (begin, _) in zip(rijen, rijen[1:]):
            if begin > einde + timedelta(days=1):
                gaten.append(
                    {"brin": brin, "na": einde.isoformat(), "voor": begin.isoformat()}
                )
    return gaten

        rapport.waarschuwingen.append(
            f"de bestanden horen bij {', '.join(namen)}; ze tellen als één bestuur."
        )
    if len(rapport.schooljaren) < _MIN_JAREN:
        rapport.waarschuwingen.append(
```

```python
"""Star van mbo-bekostiging-bestanden → 1CHO-vorm voor staat1cho.

Eén functie per vertaalregel; ``naar_1cho`` zet ze achter elkaar.
"""

from pathlib import Path

import polars as pl

KOLOMMEN_1CHO = [
    "persoonsgebonden_nummer",
    "soort_hoger_onderwijs",
    "soort_inschrijving_actuele_instelling",
    "verblijfsjaar_actuele_instelling",
    "verblijfsjaar_actuele_opleiding_instelling",
    "inschrijvingsjaar",
    "diplomajaar",
    "soort_diploma_instelling",
    "geslacht",
    "opleidingsvorm",
    "indicatie_internationale_student",
    "indicatie_eer_actueel",
    "croho_onderdeel_actuele_opleiding",
    "vestigingsnummer_gemeentenaam_volgens_rio",
    "opleiding_actueel_equivalent",
    "opleidingscode_naam_opleiding",
    "type_hoger_onderwijs_binnen_soort_hoger_onderwijs",
    "leeftijd_per_peildatum_1_oktober",
    "postcodecijfers_student_op_1_oktober",
    "postcodecijfers_van_de_hoogste_vooropl_voor_het_ho",
]
SOORT_ONDERWIJS = "mbo"
LABEL_HOOFD = "hoofdinschrijving binnen het domein actuele instelling"
LABEL_NEVEN = "neveninschrijving binnen het domein actuele instelling"
LABEL_DIPLOMA = "Diploma mbo binnen de actuele instelling"
_GESLACHT = {"M": "man", "V": "vrouw"}
_P = "_persoon_id"


def eerste_peiljaar(datum: pl.Expr) -> pl.Expr:
    """Schooljaar van de eerste 1 oktober op of na ``datum``."""
    jaar = datum.dt.year()
    return pl.when(datum <= pl.date(jaar, 10, 1)).then(jaar).otherwise(jaar + 1)


def diplomajaar(datum: pl.Expr) -> pl.Expr:
    """Schooljaar (oktober–september) waarin ``datum`` valt."""
    jaar = datum.dt.year()
    return pl.when(datum.dt.month() >= 10).then(jaar).otherwise(jaar - 1)


def leeftijd_op_1_oktober(geboortedatum: pl.Expr, jaar: pl.Expr) -> pl.Expr:
    na_peildatum = (geboortedatum.dt.month() > 10) | (
        (geboortedatum.dt.month() == 10) & (geboortedatum.dt.day() > 1)
    )
    return jaar - geboortedatum.dt.year() - na_peildatum.cast(pl.Int32)


def _niveau_getal(niveau: pl.Expr) -> pl.Expr:
    return niveau.str.extract(r"(\d)").cast(pl.Int32)


def _kies_rij_per_opleiding(sj: pl.DataFrame) -> pl.DataFrame:
    """Eén rij per persoon × schooljaar × opleiding; hoofdinschrijving wint."""
    return sj.sort(
        [_P, "Schooljaar", "Opleidingcode", "_hoofdinschrijving", "DatumBegin"],
        descending=[False, False, False, True, True],
    ).unique([_P, "Schooljaar", "Opleidingcode"], keep="first", maintain_order=True)


def _markeer_hoofdinschrijving(df: pl.DataFrame) -> pl.DataFrame:
    """Per persoon × schooljaar over het hele bestuur één hoofdinschrijving."""
    volgorde = df.with_columns(_niveau=_niveau_getal(pl.col("Niveau"))).sort(
        [
            _P,
            "Schooljaar",
            "_hoofdinschrijving",
            "_niveau",
            "Opleidingcode",
            "DatumBegin",
        ],
        descending=[False, False, True, True, False, True],
        nulls_last=True,
    )
    eerste = pl.int_range(pl.len()).over([_P, "Schooljaar"]) == 0
    return volgorde.with_columns(_hoofd=eerste & pl.col("_hoofdinschrijving")).drop(
        "_niveau"
    )


def _starts(star: dict[str, pl.DataFrame]) -> tuple[pl.DataFrame, pl.DataFrame]:
    """Eerste peiljaar per persoon en per persoon × opleiding."""
    ins = star["fact_inschrijving"]
    sj = star["fact_inschrijving_schooljaar"]
    datums = pl.concat(
        [
            ins.select(
                _P,
                "Opleidingcode",
                pl.col("DatumInschrijving").alias("_instelling"),
                pl.col("DatumBegin").alias("_opleiding"),
            ),
            sj.select(
                _P,
                "Opleidingcode",
                pl.col("DatumBegin").alias("_instelling"),
                pl.col("DatumBegin").alias("_opleiding"),
            ),
        ],
        how="vertical_relaxed",
    )
    persoon = datums.group_by(_P).agg(
        _start_instelling=eerste_peiljaar(pl.col("_instelling")).min()
    )
    opleiding = datums.group_by(_P, "Opleidingcode").agg(
        _start_opleiding=eerste_peiljaar(pl.col("_opleiding")).min()
    )
    return persoon, opleiding


def _diplomas(df: pl.DataFrame, ins: pl.DataFrame) -> pl.DataFrame:
    """Zet diplomajaar en soort diploma op de juiste rij."""
    dip = (
        ins.filter(pl.col("DIP_DatumResultaat").is_not_null())
        .select(
            _P,
            pl.col("DIP_Opleidingcode").alias("Opleidingcode"),
            diplomajaar(pl.col("DIP_DatumResultaat")).alias("_diplomajaar"),
        )
        .group_by(_P, "Opleidingcode")
        .agg(pl.col("_diplomajaar").min())
    )
    kandidaat = df.select(_P, "Opleidingcode", "Schooljaar").join(
        dip, on=[_P, "Opleidingcode"]
    )
    doel = (
        kandidaat.group_by(_P, "Opleidingcode", "_diplomajaar")
        .agg(
            pl.when(pl.col("Schooljaar") <= pl.col("_diplomajaar"))
            .then(pl.col("Schooljaar"))
            .max()
            .alias("_voor"),
            pl.col("Schooljaar").min().alias("_eerste"),
        )
        .select(
            _P,
            "Opleidingcode",
            "_diplomajaar",
            pl.coalesce("_voor", "_eerste").alias("Schooljaar"),
        )
    )
    return df.join(doel, on=[_P, "Opleidingcode", "Schooljaar"], how="left")


def naar_1cho(star: dict[str, pl.DataFrame]) -> pl.DataFrame:
    """Zet de star om naar één rij per student × inschrijvingsjaar × opleiding."""
    df = _markeer_hoofdinschrijving(
        _kies_rij_per_opleiding(star["fact_inschrijving_schooljaar"])
    )
    persoon, opleiding = _starts(star)
    df = (
        df.join(persoon, on=_P, how="left")
        .join(opleiding, on=[_P, "Opleidingcode"], how="left")
        .join(
            star["dim_deelnemer"].select(_P, "Geboortedatum", "Geslacht"),
            on=_P,
            how="left",
        )
        .join(
            star["dim_opleiding"]
            .select("Opleidingcode", "Opleiding_naam", "Opleiding_domein")
            .unique("Opleidingcode"),
            on="Opleidingcode",
            how="left",
        )
        .join(
            star["dim_instelling"].select("BRIN", "Instelling_naam").unique("BRIN"),
            on="BRIN",
            how="left",
        )
    )
    df = _diplomas(df, star["fact_inschrijving"])
    leeg = pl.lit(None, dtype=pl.Utf8)
    return df.select(
        pl.col(_P).alias("persoonsgebonden_nummer"),
        pl.lit(SOORT_ONDERWIJS).alias("soort_hoger_onderwijs"),
        pl.when(pl.col("_hoofd"))
        .then(pl.lit(LABEL_HOOFD))
        .otherwise(pl.lit(LABEL_NEVEN))
        .alias("soort_inschrijving_actuele_instelling"),
        (pl.col("Schooljaar") - pl.col("_start_instelling") + 1)
        .clip(lower_bound=1)
        .cast(pl.Int32)
        .alias("verblijfsjaar_actuele_instelling"),
        (pl.col("Schooljaar") - pl.col("_start_opleiding") + 1)
```

```python
"""R aanroepen voor staat1cho (berekenen en dashboard)."""

import os
import shutil
import subprocess
import sys
from pathlib import Path

from staat_mbo.naar_1cho import LABEL_DIPLOMA, SOORT_ONDERWIJS

MIN_STAAT1CHO = (0, 3, 0)
_WINDOWS_R = "C:/Program Files/R"
_INSTALLATIE = (
    'Installeer in R: install.packages("remotes"); '
    'remotes::install_github("cedanl/staat-van-onderwijsinstelling")'
)


class RFout(Exception):
    """R of staat1cho ontbreekt."""


def script_pad(naam: str) -> Path:
    return Path(__file__).parent / "r" / naam


def r_env() -> dict[str, str]:
    return {
        **os.environ,
        "STAAT_MBO_PYTHON": sys.executable,
        "STAAT_MBO_DIPLOMA": LABEL_DIPLOMA,
        "STAAT_MBO_SOORT": SOORT_ONDERWIJS,
    }


def vind_rscript() -> Path:
    gevonden = shutil.which("Rscript")
    if gevonden:
        return Path(gevonden)
    kandidaten = sorted(Path(_WINDOWS_R).glob("R-*/bin/Rscript.exe"))
    if kandidaten:
        return kandidaten[-1]
    raise RFout("R is niet gevonden. Installeer R: https://cran.r-project.org")


def controleer_staat1cho(rscript: Path) -> None:
    uit = subprocess.run(
        [
            str(rscript),
            "-e",
            'cat(if (requireNamespace("staat1cho", quietly = TRUE)) '
            'as.character(packageVersion("staat1cho")) else "")',
        ],
        capture_output=True,
        text=True,
    )
    versie = uit.stdout.strip()
    if not versie:
        raise RFout(f"Het R-pakket staat1cho ontbreekt. {_INSTALLATIE}")
    if tuple(int(d) for d in versie.split(".")[:3]) < MIN_STAAT1CHO:
        raise RFout(f"staat1cho {versie} is te oud (minimaal 0.3.0). {_INSTALLATIE}")


def draai_r(script: str, *args: str) -> int:
    rscript = vind_rscript()
    controleer_staat1cho(rscript)
    return subprocess.run(
        [str(rscript), str(script_pad(script)), *args], env=r_env()
    ).returncode
```

```r
## Dashboard met mbo-voorstap: bronbestanden uploaden → samenvoegen → dashboard.
## smbo_samenvoegen() roept 'python -m staat_mbo samenvoegen' aan.

smbo_samenvoegen <- function(bestanden) {
  map <- tempfile("smbo_")
  invoer <- file.path(map, "invoer")
  uitvoer <- file.path(map, "uitvoer")
  dir.create(invoer, recursive = TRUE)
  ## Shiny noemt uploads 0.csv, 1.csv; het bestandstype hangt af van de naam.
  doelen <- file.path(invoer, basename(bestanden$name))
  file.copy(bestanden$datapath, doelen)
  log <- suppressWarnings(system2(
    Sys.getenv("STAAT_MBO_PYTHON"),
    c("-m", "staat_mbo", "samenvoegen", shQuote(doelen), "--uitvoer", shQuote(uitvoer)),
    stdout = TRUE, stderr = TRUE
  ))
  status <- attr(log, "status")
  meldingen_pad <- file.path(uitvoer, "meldingen.txt")
  meldingen <- if (file.exists(meldingen_pad)) {
    readLines(meldingen_pad, encoding = "UTF-8")
  } else {
    c("Niet gelukt.", utils::tail(log, 5))
  }
  list(
    ok = is.null(status) || status == 0,
    pad = file.path(uitvoer, "1cho_mbo.csv"),
    meldingen = meldingen
  )
}

if (Sys.getenv("STAAT_MBO_ALLEEN_FUNCTIES") != "1") {
```
